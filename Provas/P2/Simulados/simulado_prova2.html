/* ── Busca ─────────────────────────────────────────────────── */
int busca_rec(int v[], int ini, int fim, int x) {
    if (ini > fim) return -1;
    int meio = (ini + fim) / 2;
    if (v[meio] == x) return meio;
    if (v[meio] < x)  return busca_rec(v, meio + 1, fim, x);
    return busca_rec(v, ini, meio - 1, x);
}

int contaOcorr(int v[], int n, int x) {
    int c = 0;
    for (int i = 0; i < n; i++)
        if (v[i] == x) c++;
    return c;
}

/* ── Ordenação ──────────────────────────────────────────────── */
void insertionSort(int v[], int n) {
    for (int i = 1; i < n; i++) {
        int chave = v[i], j = i - 1;
        while (j >= 0 && v[j] > chave) {
            v[j+1] = v[j];
            j--;
        }
        v[j+1] = chave;
    }
}

int separa(int v[], int ini, int fim) {
    int pivo = v[fim], i = ini - 1;
    for (int j = ini; j < fim; j++) {
        if (v[j] <= pivo) {
            i++;
            int tmp = v[i]; v[i] = v[j]; v[j] = tmp;
        }
    }
    int tmp = v[i+1]; v[i+1] = v[fim]; v[fim] = tmp;
    return i + 1;
}

/* ── Complexidade ───────────────────────────────────────────── */
int f3(int n) {
    int c = 0;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j += 2)

    return c;
}

int f4(int n) {

}

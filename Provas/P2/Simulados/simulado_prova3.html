/* ── Q3 · Busca binária em nome ordenado, conta iterações ───────── */
int buscaNome(char *v, int n, char c) {
    int ini = 0, fim = n - 1, pcont = 0;
    while (ini <= fim) {
        pcont++;
        int meio = (ini + fim) / 2;
        if (v[meio] == c)      break;
        else if (c < v[meio])  fim  = meio - 1;
        else                   ini  = meio + 1;
    }
    return pcont;
}

/* ── Q4 · Ordenação decrescente por seleção em string ───────────── */
int sortDesc(char *vet, int n) {
    int pc = 0; char aux;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++)
            if (vet[i] < vet[j]) { aux=vet[i]; vet[i]=vet[j]; vet[j]=aux; pc++; }
    return pc;
}

/* ── Q5 · Insertion sort em string, conta deslocamentos ─────────── */
int insertionStr(char *vet, int n) {
    int trocas = 0; char chave;
    for (int i = 1; i < n; i++) {
        chave = vet[i]; int j = i - 1;
        while (j >= 0 && vet[j] > chave) { vet[j+1] = vet[j]; j--; trocas++; }
        vet[j+1] = chave;
    }
    return trocas;
}

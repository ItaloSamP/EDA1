/* ── Q1 · Conta chars com ASCII maior que um limite ────────────── */
int conta_maiores(char *s, char limite) {
    if (*s == '\0') return 0;
    return (*s > limite ? 1 : 0) + conta_maiores(s + 1, limite);
}

/* ── Q2 · Primeira posição de char derivado de índice ───────────── */
int primeiraPos(char *s, char c) {
    for (int i = 0; s[i] != '\0'; i++)
        if (s[i] == c) return i;
    return -1;
}

/* ── Q3 · Busca binária em nome ordenado, conta iterações ───────── */
int buscaNome(char *v, int n, char c) {
    int ini = 0, fim = n - 1, pcont = 0;
    while (ini <= fim) {
        pcont++;
        int meio = (ini + fim) / 2;
        if (v[meio] == c)      break;
        else if (c < v[meio])  fim  = meio - 1;
        else                   ini  = meio + 1;
    }
    return pcont;
}

/* ── Q4 · Bubble sort (1ª passagem) em string, conta trocas ─────── */
int bubblePasseStr(char *vet, int n) {
    int trocas = 0; char aux;
    for (int j = 0; j < n - 1; j++)
        if (vet[j] > vet[j+1]) { aux=vet[j]; vet[j]=vet[j+1]; vet[j+1]=aux; trocas++; }
    return trocas;
}

/* ── Q5 · Selection sort crescente em string, conta trocas ──────── */
int minIdxStr(char *v, int ini, int n) {
    int m = ini;
    for (int i = ini + 1; i < n; i++) if (v[i] < v[m]) m = i;
    return m;
}
int selSortStr(char *vet, int n) {
    int trocas = 0; char aux;
    for (int i = 0; i < n - 1; i++) {
        int m = minIdxStr(vet, i, n);
        if (m != i) { aux=vet[i]; vet[i]=vet[m]; vet[m]=aux; trocas++; }
    }
    return trocas;
}

/* ── Q6 · Lista: conta nós com valor ≥ x ───────────────────────── */
typedef struct No { int val; struct No *prox; } No;
int contaMaiorIgual(No *l, int x) {
    int c = 0;
    while (l) { if (l->val >= x) c++; l = l->prox; }
    return c;
}

/* ── Q7 · Recursão: conta char em string completa ───────────────── */
int contaChar(char *s, char c) {
    if (*s == '\0') return 0;
    return (*s == c ? 1 : 0) + contaChar(s + 1, c);
}

/* ── Q8 · Complexidade: loop com passo variável por linha ──────── */
int p(int n) {
    int c = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j += i)
            c++;
    return c;
}

/* ── Q9 · Lista: k-ésimo a partir do final (dois ponteiros) ─────── */
No* coda(No *l, int k) {
    No *ini = l, *fim = l;
    for (int i = 0; i < k; i++) fim = fim->prox;
    while (fim != NULL) { ini = ini->prox; fim = fim->prox; }
    return ini;
}

/* ── Q10 · MDC de Euclides com valores derivados da lista ─────── */
int soma(No *l)    { int s=0; while(l){s+=l->val; l=l->prox;} return s; }
int tamanho(No *l) { int c=0; while(l){c++; l=l->prox;} return c; }
int mdc(int a, int b) {
    while (b != 0) { int r = a % b; a = b; b = r; }
    return a;

/* ── Busca ─────────────────────────────────────────────────── */
int busca_seq(int v[], int n, int x) {
    for (int i = 0; i < n; i++)
        if (v[i] == x) return i;
    return -1;
}

int busca_bin(int v[], int n, int x) {
    int ini = 0, fim = n - 1, meio;
    while (ini <= fim) {
        meio = (ini + fim) / 2;
        if (v[meio] == x) return meio;
        if (v[meio] < x)  ini = meio + 1;
        else               fim = meio - 1;
    }
    return -1;
}

/* ── Ordenação ──────────────────────────────────────────────── */
int bubblePasse1(int v[], int n) {
    int trocas = 0;
    for (int j = 0; j < n - 1; j++)
        if (v[j] > v[j+1]) {
            int tmp = v[j]; v[j] = v[j+1]; v[j+1] = tmp;
            trocas++;
        }
    return trocas;
}

int minIdx(int v[], int ini, int n) {
    int m = ini;
    for (int i = ini + 1; i < n; i++)
        if (v[i] < v[m]) m = i;
    return m;
}

int selTrocas(int v[], int n) {
    int trocas = 0;
    for (int i = 0; i < n - 1; i++) {
        int m = minIdx(v, i, n);
        if (m != i) {
            int tmp = v[i]; v[i] = v[m]; v[m] = tmp;
            trocas++;
        }
    }
    return trocas;
}

/* ── Complexidade ───────────────────────────────────────────── */
int f1(int n) {
    int c = 0;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++)
            c++;
    return c;
}

int f2(int n) {
    int c = 0;
    for (int i = 1; i <= n; i *= 2)
        c++;
    return c;
}

int tamanho(No* l) {

    int s = 0;

    return s;
}

int maximo(No* l) {
    int m = l->val; l = l->prox;
    while (l) { if (l->val > m) m = l->val; l = l->prox; }
    return m;
}

int contaMaior(No* l, int x) {
    int c = 0;
    while (l) { if (l->val > x) c++; l = l->prox; }
    return c;
}

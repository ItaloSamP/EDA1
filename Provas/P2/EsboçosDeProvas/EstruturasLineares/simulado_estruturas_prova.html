typedef struct No { int dado; struct No *prox; } No;

/* Lista: 2 → 3 → 2 → 0 → 3 → 7 → 9 → 3 → 7 → NULL   (n=9)
          dígitos de 232037937 na ordem original         */

/* ── Funções ────────────────────────────────────────────────────── */
int tamanho(No *l) {
    int c = 0;
    for (No *p = l; p; p = p->prox) c++;
    return c;
}

int nth(No *l, int k) {              /* elemento no índice k (base 0) */
    for (int i = 0; l && i < k; i++) l = l->prox;
    return l ? l->dado : -1;
}

int soma(No *l) {
    int s = 0;
    for (No *p = l; p; p = p->prox) s += p->dado;
    return s;
}

int busca(No *l, int x) {            /* posição da 1ª ocorrência      */
    int pos = 0;
    for (No *p = l; p; p = p->prox, pos++)
        if (p->dado == x) return pos;
    return -1;
}

int maximo(No *l) {
    int m = l->dado;
    for (No *p = l->prox; p; p = p->prox)
        if (p->dado > m) m = p->dado;
    return m;
}

int contaMaior(No *l, int x) {       /* nós com dado > x              */
    int c = 0;
    for (No *p = l; p; p = p->prox)
        if (p->dado > x) c++;
    return c;
}

int penultimo(No *l) {               /* dado do nó penúltimo          */
    if (!l || !l->prox) return -1;
    while (l->prox->prox) l = l->prox;
    return l->dado;
}

int somaPos(No *l) {                 /* soma dos dados em pos pares (0,2,4,…) */
    int s = 0, i = 0;
    for (No *p = l; p; p = p->prox, i++)
        if (i % 2 == 0) s += p->dado;
    return s;
}

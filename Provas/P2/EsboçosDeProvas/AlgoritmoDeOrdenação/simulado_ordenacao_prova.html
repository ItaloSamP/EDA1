/* ── Selection Sort ─────────────────────────────────────────────── */
int minIdx(int v[], int esq, int n) {
    int m = esq;
    for (int j = esq+1; j < n; j++)
        if (v[j] < v[m]) m = j;
    return m;
}

int selTrocas(int v[], int n) {             /* retorna total de trocas   */
    int trocas = 0;
    for (int i = 0; i < n-1; i++) {
        int m = minIdx(v, i, n);
        if (m != i) {
            int tmp = v[i]; v[i] = v[m]; v[m] = tmp;
            trocas++;
        }
    }
    return trocas;
}

/* ── Bubble Sort ────────────────────────────────────────────────── */
int bubblePasse1(int v[], int n) {          /* trocas na 1ª passagem     */
    int trocas = 0;
    for (int j = 0; j < n-1; j++) {
        if (v[j] > v[j+1]) {
            int tmp = v[j]; v[j] = v[j+1]; v[j+1] = tmp;
            trocas++;
        }
    }
    return trocas;
}

/* ── QuickSort — partição ───────────────────────────────────────── */
int separa(int v[], int esq, int dir) {
    int piv = v[dir], i = esq - 1, tmp;
    for (int j = esq; j < dir; j++) {
        if (v[j] <= piv) {
            i++;
            tmp = v[i]; v[i] = v[j]; v[j] = tmp;
        }
    }
    tmp = v[i+1]; v[i+1] = v[dir]; v[dir] = tmp;
    return i + 1;
}

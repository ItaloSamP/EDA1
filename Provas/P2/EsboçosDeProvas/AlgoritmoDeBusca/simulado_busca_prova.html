int busca1(int v[], int n, int x) {          /* sequencial                */
    int i;
    for (i = 0; i < n && v[i] != x; i++);
    return (i < n) ? i : -1;
}

int busca2(int v[], int n, int x) {          /* binária iterativa         */
    int esq = 0, dir = n - 1;
    while (esq <= dir) {
        int meio = (esq + dir) / 2;
        if      (v[meio] == x) return meio;
        else if (v[meio] <  x) esq = meio + 1;
        else                   dir = meio - 1;
    }
    return -1;
}

int busca3(int v[], int n, int x) {          /* seq em vetor ordenado     */
    for (int i = 0; i < n; i++) {
        if (v[i] > x) return -1;
        if (v[i] == x) return i;
    }
    return -1;
}

int busca4(char *s, char c) {                /* 1ª ocorrência de char     */
    for (int i = 0; s[i]; i++)
        if (s[i] == c) return i;
    return -1;
}

int busca5(int v[], int n, int x) {          /* binária — conta passos    */
    int esq = 0, dir = n - 1, passos = 0;
    while (esq <= dir) {
        passos++;
        int meio = (esq + dir) / 2;
        if      (v[meio] == x) return passos;
        else if (v[meio] <  x) esq = meio + 1;
        else                   dir = meio - 1;
    }
    return -1;
}

int busca6(int v[], int i, int n, int x) {   /* recursiva a partir de i  */
    if (i >= n)    return -1;
    if (v[i] == x) return i;
    return busca6(v, i + 1, n, x);
}

int conta(int v[], int n, int x) {           /* conta ocorrências de x   */
    int c = 0;
    for (int i = 0; i < n; i++) c += (v[i] == x);
    return c;
}

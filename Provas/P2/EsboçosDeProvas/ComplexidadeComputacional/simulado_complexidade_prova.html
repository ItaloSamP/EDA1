    return c;

int f2(int n) {                      /* pares (i,j) com j>i           */
    int s = 0;
    for (int i = 0; i < n; i++)
        for (int j = i+1; j < n; j++)
            s++;
    return s;
}

int f3(int n) {                      /* iterações: i=1; i*=2; i<=n   */
    int s = 0;
    for (int i = 1; i <= n; i *= 2)
        s++;
    return s;
}

int f4(int n) {                      /* profundidade da recursão /2   */
    if (n <= 1) return 1;
    return f4(n / 2) + 1;
}

int f5(int n) {                      /* soma geométrica n+n/2+n/4+…   */
    int s = 0;
    for (int i = n; i > 0; i /= 2)
        s += i;
    return s;
}

int f6(int n) {                      /* laço: outer×inner(j*j<=n)    */
    int s = 0;
    for (int i = 0; i < n; i++)
        for (int j = 1; j*j <= n; j++)
            s++;
    return s;
}
